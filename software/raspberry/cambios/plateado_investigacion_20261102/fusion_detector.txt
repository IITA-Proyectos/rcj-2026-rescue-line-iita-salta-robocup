from __future__ import annotations

import os
import threading
import time

import cv2
import numpy as np

try:
    import torch
    import torch.nn as nn
except Exception as e:
    torch = None
    nn = None
    _TORCH_IMPORT_ERROR = e


class SilverLineDetector:
    def __init__(self, model_path: str, device: str = "cpu", input_size: int = 64, num_threads: int = 2):
        if torch is None:
            raise ImportError(f"torch is not available: {_TORCH_IMPORT_ERROR}")

        self.model_path = model_path
        self.input_size = int(input_size)
        self.device = torch.device(device)

        torch.set_num_threads(int(num_threads))

        self.model = self._load_model(model_path)
        self.model.to(self.device)
        self.model.eval()

        self.class_names = ["No Silver", "Silver"]

        self._mean = np.array([0.485, 0.456, 0.406], dtype=np.float32).reshape(1, 1, 3)
        self._std = np.array([0.229, 0.224, 0.225], dtype=np.float32).reshape(1, 1, 3)

        self._model_lock = threading.Lock()
        self._frame_lock = threading.Lock()
        self._result_lock = threading.Lock()
        self._wake_event = threading.Event()

        self._latest_frame = None
        self._latest_frame_time = 0.0
        self._latest_result = None

        self._running = True
        self._warmup()

        self._worker = threading.Thread(target=self._worker_loop, daemon=True)
        self._worker.start()

    def _load_model(self, model_path: str):
        if not os.path.exists(model_path):
            raise FileNotFoundError(f"Silver model not found: {model_path}")

        try:
            return torch.jit.load(model_path, map_location=self.device)
        except Exception:
            pass

        try:
            return torch.load(model_path, map_location=self.device)
        except Exception as e:
            raise RuntimeError(f"Failed to load model: {model_path} ({e})")

    def _warmup(self) -> None:
        x = torch.randn(1, 3, self.input_size, self.input_size, device=self.device)

        with self._model_lock:
            with torch.inference_mode():
                self.model(x)

    def _preprocess(self, rgb: np.ndarray) -> "torch.Tensor":
        img = cv2.resize(rgb, (self.input_size, self.input_size), interpolation=cv2.INTER_AREA)
        img = img.astype(np.float32) / 255.0
        img = (img - self._mean) / self._std

        x = torch.from_numpy(img).to(self.device)
        return x.permute(2, 0, 1).unsqueeze(0)

    def predict(self, rgb: np.ndarray) -> dict:
        x = self._preprocess(rgb)

        with self._model_lock:
            with torch.inference_mode():
                out = self.model(x)
                probs = torch.softmax(out, dim=1)
                pred = int(torch.argmax(out, dim=1).item())
                conf = float(probs[0, pred].item())

        return {
            "prediction": pred,
            "class_name": self.class_names[pred] if pred < len(self.class_names) else str(pred),
            "confidence": conf,
        }

    def submit(self, rgb: np.ndarray) -> bool:
        if rgb is None or not self._running:
            return False

        with self._frame_lock:
            self._latest_frame = rgb.copy()
            self._latest_frame_time = time.perf_counter()
            self._wake_event.set()

        return True

    def poll_result(self):
        with self._result_lock:
            result = self._latest_result
            self._latest_result = None
            return result

    def _worker_loop(self) -> None:
        while self._running:
            self._wake_event.wait()

            if not self._running:
                break

            while self._running:
                with self._frame_lock:
                    if self._latest_frame is None:
                        self._wake_event.clear()
                        break

                    frame = self._latest_frame
                    captured_at = self._latest_frame_time
                    self._latest_frame = None

                try:
                    inference_start = time.perf_counter()
                    result = self.predict(frame)
                    inference_end = time.perf_counter()

                    result["captured_at"] = captured_at
                    result["completed_at"] = inference_end
                    result["inference_ms"] = (inference_end - inference_start) * 1000

                    with self._result_lock:
                        self._latest_result = result

                except Exception as e:
                    print(f"Silver inference error: {e}")

    def close(self) -> None:
        self._running = False
        self._wake_event.set()
        self._worker.join(timeout=1.0)
